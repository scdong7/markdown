create procedure p_etl_log_detail(in v_type varchar(200),in v_username varchar(10),in v_procname varchar(50),in v_datadate varchar(10),in v_comments varchar(1000),in v_status varchar(20),in v_starttime datetime,in v_endtime datetime,in v_sqlcode varchar(100),in v_sqlerror varchar(3000),in v_dealrow integer)
begin
	if v_type='0' then 
		insert into etl_log_detail(
		user_name,proc_name,data_date,comments,status,start_time)
		value(
		v_username,v_procname,v_datadate,v_comments,'Running',v_starttime);
	else 
		update etl_log_detail t
		set t.status=v_status,
		    t.sql_code=v_sqlcode,
		    t.sql_error=v_sqlerror,
		    t.end_time=v_endtime,
		    t.run_times=timestampdiff(SECOND,start_time,v_endtime),
		    t.deal_row=v_dealrow
		where t.user_name=v_username and t.proc_name=v_procname and t.data_date=v_datadate and t.status='Running';
	end if;
	commit;
end;

create table etl_log_detail(
user_name varchar(100),
proc_name varchar(200),
data_date varchar(10),
comments varchar(4000),
sql_code varchar(100),
sql_error varchar(10000),
status varchar(20),
start_time datetime,
end_time datetime,
run_times integer,
deal_row integer
);
